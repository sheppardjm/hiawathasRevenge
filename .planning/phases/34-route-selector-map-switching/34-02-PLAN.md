**3. RouteSelector L.Control (add to initMap, after ResetControl):**

This is the CANONICAL click handler — one version that handles aria-checked, tabindex, background color, AND renderRoute call:

```javascript
const RouteSelector = L.Control.extend({
  options: { position: 'topleft' },
  onAdd() {
    const container = L.DomUtil.create('div', 'leaflet-bar leaflet-control route-selector');
    container.setAttribute('role', 'radiogroup');
    container.setAttribute('aria-label', 'Select route distance');
    L.DomEvent.disableClickPropagation(container);

    const routes = routesManifest.routes;
    const btns = [];

    routes.forEach((route, i) => {
      const btn = L.DomUtil.create('button', 'route-selector__btn', container);
      btn.textContent = route.shortName;   // "100mi", "100k", "50k"
      btn.dataset.routeId = route.id;
      btn.setAttribute('role', 'radio');
      btn.setAttribute('aria-checked', i === 0 ? 'true' : 'false');
      btn.setAttribute('tabindex', i === 0 ? '0' : '-1');

      // Set initial active styling for the first button
      if (i === 0) {
        btn.style.background = route.color;
        btn.style.color = forest900;
      }

      btns.push(btn);

      // Single canonical click handler — updates ALL state: aria, tabindex, styles, route
      L.DomEvent.on(btn, 'click', () => {
        if (route.id === activeRouteId) return;  // no-op same route

        // Update all buttons: aria-checked, tabindex, and background styles
        btns.forEach(b => {
          const isActive = b.dataset.routeId === route.id;
          b.setAttribute('aria-checked', isActive ? 'true' : 'false');
          b.setAttribute('tabindex', isActive ? '0' : '-1');
          b.style.background = isActive ? route.color : 'transparent';
          b.style.color = isActive ? forest900 : '';
        });

        renderRoute(route.id);
      });
    });

    // Arrow key navigation — roving tabindex (W3C APG radiogroup)
    L.DomEvent.on(container, 'keydown', (e) => {
      const currentIdx = btns.findIndex(b => b === document.activeElement);
      if (currentIdx === -1) return;
      let nextIdx = -1;
      if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
        L.DomEvent.preventDefault(e);
        L.DomEvent.stopPropagation(e);
        nextIdx = (currentIdx + 1) % btns.length;
      } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
        L.DomEvent.preventDefault(e);
        L.DomEvent.stopPropagation(e);
        nextIdx = (currentIdx - 1 + btns.length) % btns.length;
      }
      if (nextIdx !== -1) {
        btns[nextIdx].focus();
        btns[nextIdx].click();  // arrow = select + focus per radiogroup pattern
      }
    });

    return container;
  }
});
```

**3. Create updateChart(routeId) function at module scope:**

```javascript
async function updateChart(routeId) {
  if (!chart) return;

  const [routeData, annotations] = await Promise.all([
    fetch(`/data/${routeId}/route-data.json`).then(r => r.json()),
    fetch(`/data/${routeId}/annotations.json`).then(r => r.json()),
  ]);

  // Replace dataset data in-place (Chart.js v4 documented pattern)
  const newData = routeData.points.map(pt => ({
    x: pt.miles,
    y: +(pt.ele * 3.28084).toFixed(1)
  }));
  chart.data.datasets[0].data = newData;

  // Update x-axis max to new route distance
  chart.options.scales.x.max = routeData.meta.totalMiles;

  // Rebuild sector band annotations (preserve crosshair)
  const SECTOR_BAND_FILL = hexToRgba(getCSSColor('--color-amber-500'), 0.18);
  const crosshair = chart.options.plugins.annotation.annotations.crosshair;
  const newAnnotations = { crosshair };
  const sectors = annotations.filter(a => a.type === 'sector');
  for (const sector of sectors) {
    newAnnotations[sector.id] = {
      type: 'box',
      xMin: sector.startMile,
      xMax: sector.endMile,
      backgroundColor: SECTOR_BAND_FILL,
      borderWidth: 0,
      drawTime: 'beforeDatasetsDraw'
    };
  }
  chart.options.plugins.annotation.annotations = newAnnotations;

  // Instant update — no animation
  chart.update('none');
}
```

```javascript
canvas.addEventListener('mousemove', (e) => {
  if (!chart) return;
  const rect = canvas.getBoundingClientRect();
  const pos = { x: e.clientX - rect.left, y: e.clientY - rect.top };
  const miles = chart.scales.x.getValueForPixel(pos.x);
  if (miles == null) return;
  const currentData = chart.data.datasets[0].data;
  const clamped = Math.max(0, Math.min(currentData[currentData.length - 1].x, miles));

});
```
