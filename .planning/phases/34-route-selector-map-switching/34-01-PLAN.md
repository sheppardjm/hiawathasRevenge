```
async function renderRoute(routeId) {
  // 1. Fetch route-specific data in parallel
  const [routeData, annotations, sectorElevations, surfacePoints] = await Promise.all([
    fetch(`/data/${routeId}/route-data.json`).then(r => r.json()),
    fetch(`/data/${routeId}/annotations.json`).then(r => r.json()),
    fetch(`/data/${routeId}/sector-elevations.json`).then(r => r.json()),
    fetch(`/data/${routeId}/surface-points.json`).then(r => r.json()),
  ]);

  // 2. Clear previous route layers
  clearActiveRoute();

  // 3. Build latlngs and update module-scope references
  const latlngs = routeData.points.map(pt => [pt.lat, pt.lon]);
  routePoints = routeData.points;

  // 4. Draw surface-colored polyline (run-flush algorithm)
  drawSurfacePolyline(latlngs, surfacePoints, activeRouteGroup);

  // 5. Build sector overlays (visible + ghost per sector)
  const sectors = annotations.filter(a => a.type === 'sector');
  for (const sector of sectors) {
    const sectorPts = latlngs.slice(sector.startIdx, sector.endIdx + 1);
    const defaultStyle = { color: SECTOR_COLOR.line, weight: 5, opacity: 0.85 };
    const visiblePoly = L.polyline(sectorPts, { ...defaultStyle, interactive: false });
    const ghostPoly = L.polyline(sectorPts, { color: SECTOR_COLOR.line, weight: 20, opacity: 0, interactive: true });
    activeRouteGroup.addLayer(visiblePoly);
    activeRouteGroup.addLayer(ghostPoly);
    const entry = { sectorId: sector.id, visiblePoly, ghostPoly, defaultStyle };
    sectorLayerEntries.push(entry);
    // Wire hover/click events (same logic as current, but using sectorDetailsData and sectorElevations)
    ghostPoly.on('mouseover', () => { if (activeSector?.sectorId !== sector.id) visiblePoly.setStyle({ weight: 7, opacity: 1.0 }); });
    ghostPoly.on('mouseout', () => { if (activeSector?.sectorId !== sector.id) visiblePoly.setStyle(defaultStyle); });
    ghostPoly.on('click', () => {
      if (activeSector && activeSector.sectorId !== sector.id) activeSector.visiblePoly.setStyle(activeSector.defaultStyle);
      activeSector = entry;
      visiblePoly.setStyle({ weight: 8, opacity: 1.0 });
      visiblePoly.bringToFront();
      const details = sectorDetailsData.find(d => d.id === sector.id);
      const elevData = sectorElevations.find(e => e.id === sector.id);
      openPanel({ sector, details, elevData });
    });
  }

  // 6. Build sector labels — add ALL to activeRouteGroup unconditionally, control visibility via opacity
  sectorLabels = [];
  for (const sector of sectors) {
    const midIdx = Math.floor((sector.startIdx + sector.endIdx) / 2);
    const midPt = latlngs[midIdx];
    const bgColor = LABEL_COLOR;
    const stars = '\u2605'.repeat(sector.stars);

    const labelIcon = L.divIcon({
      className: 'sector-label',
      html: `<div style="
        background: ${bgColor};
        color: ${labelTextColor};
        border: 2px solid ${labelTextColor};
        border-radius: 12px;
        padding: 3px 8px;
        font-family: var(--font-display);
        font-size: 11px;
        font-weight: 700;
        white-space: nowrap;
        transform: translate(-50%, -50%);
        box-shadow: 2px 2px 0px rgba(0,0,0,0.4);
        line-height: 1.3;
        text-align: center;
      ">${sector.name}<br><span style="font-size: 9px; letter-spacing: 1px;">${stars}</span></div>`,
      iconSize: [0, 0],
      iconAnchor: [0, 0],
    });

    const labelMarker = L.marker(midPt, {
      icon: labelIcon,
      interactive: false,
      keyboard: false,
      zIndexOffset: 250,
    });

    // Add to activeRouteGroup unconditionally — zoom gating uses opacity, not add/remove
    activeRouteGroup.addLayer(labelMarker);
    sectorLabels.push(labelMarker);
  }
  updateLabelVisibility(); // apply zoom gate to new labels

  // 7. Build restock markers — clickable water-drop icons with name/mileage popups
  const restocks = annotations.filter(a => a.type === 'restock');
  for (const stop of restocks) {
    const restockIcon = L.divIcon({
      className: 'restock-marker',
      html: `<svg xmlns="http://www.w3.org/2000/svg" width="20" height="26" viewBox="0 0 20 26">
    <path d="M10 0C10 0 0 12 0 17c0 5.523 4.477 10 10 10s10-4.477 10-10C20 12 10 0 10 0z" fill="${lake400}" stroke="${forest900}" stroke-width="1.5"/>
    <circle cx="10" cy="17" r="3" fill="rgba(255,255,255,0.5)"/>
  </svg>`,
      iconSize: [20, 26],
      iconAnchor: [10, 26],
      popupAnchor: [0, -26]
    });

    const restockMarker = L.marker([stop.lat, stop.lon], {
      icon: restockIcon,
      title: stop.name,
      zIndexOffset: 500
    }).bindPopup(
      `<strong>${stop.name}</strong><br>Mile ${stop.mile}`,
      { className: 'restock-popup' }
    );

    activeRouteGroup.addLayer(restockMarker);
  }

  // 8. Update bounds and fit
  initialBounds = L.latLngBounds(latlngs);
  map.fitBounds(initialBounds, { padding: [20, 20], animate: !prefersReducedMotion });

  // 9. Panel close-on-switch: close if active sector not on new route
  const routeConfig = routesManifest.routes.find(r => r.id === routeId);
  if (activeSector && !routeConfig.sectorIds.includes(activeSector.sectorId)) {
    closePanel();
  }

  // 10. Update active route tracking
  activeRouteId = routeId;
}
```

**Create clearActiveRoute() function** (inside initMap closure):

```
function clearActiveRoute() {
  if (activeSector) {
    activeSector.visiblePoly.setStyle(activeSector.defaultStyle);
    activeSector = null;
  }
  if (activeRouteGroup) {
    activeRouteGroup.clearLayers();
  }
  sectorLayerEntries.length = 0;
  sectorLabels = [];
}
```

**Create drawSurfacePolyline(latlngs, surfacePoints, layerGroup) function** (inside initMap closure):
```
function drawSurfacePolyline(latlngs, surfacePoints, layerGroup) {
  // Assert alignment
  if (surfacePoints.length !== latlngs.length) {
    console.warn(`Surface points (${surfacePoints.length}) != route points (${latlngs.length}), falling back to uniform color`);
    layerGroup.addLayer(L.polyline(latlngs, { color: forest900, weight: 4, opacity: 0.9, smoothFactor: 1, interactive: false }));
    return;
  }
  // Run-flush algorithm
  let prev = null;
  let runStart = 0;
  function flushRun(endIdx) {
    if (prev === null) return;
    const pts = latlngs.slice(runStart, endIdx + 1);
    if (pts.length >= 2) {
      layerGroup.addLayer(L.polyline(pts, {
        color: SURFACE_COLORS[prev] || SURFACE_COLORS.unknown,
        weight: 4, opacity: 0.9, smoothFactor: 1, interactive: false,
      }));
    }
  }
  for (let i = 0; i < surfacePoints.length; i++) {
    const s = surfacePoints[i].surface;
    if (s !== prev) { flushRun(i); runStart = i; prev = s; }
  }
  flushRun(surfacePoints.length - 1);
}
```

**Refactor updateLabelVisibility()** to use opacity instead of add/remove:

Labels are always in activeRouteGroup (added unconditionally in step 6 of renderRoute). Zoom gating controls visibility via opacity on the label marker's underlying DOM element. This avoids conflicts with clearLayers() — when activeRouteGroup.clearLayers() runs on route switch, all labels are removed cleanly because they are always in the group.

```
function updateLabelVisibility() {
  const zoom = map.getZoom();
  for (const marker of sectorLabels) {
    const el = marker.getElement?.();
    if (!el) continue;
    el.style.opacity = zoom >= 12 ? '1' : '0';
    el.style.pointerEvents = zoom >= 12 ? 'auto' : 'none';
  }
}
map.on('zoomend', updateLabelVisibility);
```

Note: `marker.getElement()` returns null if the marker has not been rendered to the DOM yet. The guard `if (!el) continue;` handles this safely. On the first call from renderRoute, Leaflet may not have flushed DOM yet — the `zoomend` handler will catch up once the map settles after fitBounds.
