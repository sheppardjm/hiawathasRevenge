    background-color: var(--color-sun-500);
    color: var(--color-forest-950);

  .gold-section :global(h3) {
    color: var(--color-forest-950);
    text-shadow: none;
  }
  .gold-section :global(.donate-message) {
    color: var(--color-forest-900);
  }
  .gold-section :global(.donate-button) {
    color: var(--color-forest-950);
    border-color: var(--color-forest-900);
    box-shadow: 4px 4px 0px rgba(0, 0, 0, 0.2);
  }
  .gold-section :global(.donate-button:hover) {
    background-color: var(--color-forest-950);
    color: var(--color-sun-300);
